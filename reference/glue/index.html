    def get_request_payload() -> dict[str, Any]:
    """Get the JSON payload from the Flask request."""
    if not flask.has_request_context():
        raise (
            RuntimeError("No Flask request context available")
        ) from None
    if flask.request.method == "GET":
        return dict(flask.request.args)

    # Bodyless requests (no Content-Type, empty body) are common for
    # DELETE calls from API clients (campus_python/campus-cli send
    # DELETE without a body), so treat them as an empty payload.
    if not flask.request.get_data(cache=True):
        return {}

    json_payload = flask.request.get_json(silent=True)
    if json_payload is None:
        raise api_errors.InvalidRequestError(
            message="Malformed JSON payload",
            error_code="MALFORMED_REQUEST",
            # Error details must stay JSON-serializable; the raw body is bytes
            body=flask.request.data.decode("utf-8", errors="replace"),
        ) from None
    if not isinstance(json_payload, dict):
        raise api_errors.InvalidRequestError(
            message="Expected object in JSON payload",
            body=json_payload,
        ) from None
    return json_payload

    def unpack_into(
        func: Callable[..., Any],
        **request_args: Any,
) -> Any:
    """Unpack request arguments into the given function's arguments,
    based on its signature.

    String values (query/path parameters) are coerced to the annotated
    int/float/bool type; a failed coercion is reported as a field error.

    Raises ValidationError with structured field errors for any issues.
    """
    reconciled, extra_args, missing_params = parameter.reconcile(
        request_args,
        func
    )

    field_errors: list[FieldError] = []

    # Check if function accepts **kwargs
    func_params = inspect.signature(func).parameters
    has_var_keyword = any(
        p.kind == inspect.Parameter.VAR_KEYWORD
        for p in func_params.values()
    )

    # Only raise error for extra arguments if function doesn't have **kwargs
    if extra_args and not has_var_keyword:
        field_errors.extend([
            FieldError(
                field=param,
                code="UNRECOGNIZED_FIELD",
                message=f"Unexpected field: {param}"
            )
            for param in extra_args
        ])

    if missing_params:
        field_errors.extend([
            FieldError(
                field=param,
                code="MISSING",
                message=f"Missing required field: {param}"
            )
            for param in missing_params
        ])

    for name, value in reconciled.items():
        param = func_params.get(name)
        if param is None or parameter.is_variadic(param):
            continue
        try:
            reconciled[name] = _coerce_str_value(value, param.annotation)
        except ValueError:
            field_errors.append(FieldError(
                field=name,
                code="INVALID_TYPE",
                message=f"Expected {param.annotation} for field: {name}"
            ))

    # Reject values whose type does not match the annotation, so that
    # e.g. a string scopes field is not silently iterated char-by-char
    for name, value in reconciled.items():
        param = func_params.get(name)
        if param is None or parameter.is_variadic(param):
            continue
        compatible = _is_type_compatible(value, param.annotation)
        if compatible is None:
            continue
        if compatible:
            reconciled[name] = _coerce_to_annotation(value, param.annotation)
        else:
            field_errors.append(FieldError(
                field=name,
                code="INVALID_TYPE",
                message=f"Invalid type for field: {name}"
            ))

    if field_errors:
        raise ValidationError(
            message="One or more fields are invalid",
            errors=field_errors
        )

    # Call the original function with unpacked arguments
    # Include extra_args only if function has **kwargs
    return func(**reconciled, **extra_args)

    def unpack_request(
        func: Callable[..., Any]
) -> Callable[[], Any]:
    """Decorator that unpacks Flask request into the decorated function's
    arguments, based on its signature.

    GET requests will use URL parameters, POST/PUT requests will use JSON body.
    """
    # Validate func annotations
    if not func.__annotations__:
        raise (
            ValueError(f"Function {func.__name__} missing type annotations")
        ) from None
    incompatible_params = [
        param for param in inspect.signature(func).parameters.values()
        if not parameter.is_keyword_supported(param)
    ]
    if incompatible_params:
        raise ValueError(
            f"Parameters {incompatible_params} must be "
            "keyword-argument-compatible"
        ) from None

    @wraps(func)
    def wrappervf(*args, **kwargs) -> Any:
        """The view function presented to Flask"""
        assert not args, f"Positional arguments not supported: {args}"
        request_args = get_request_payload()
        return unpack_into(func, **kwargs, **request_args)

    return wrappervf

    def validate_request_and_extract_json(
        schema: Mapping[str, Type], *,
        on_error: types.ErrorHandler,
) -> types.JsonObject:
    """Validate the request JSON body against the provided schema before
    returning the payload.
    """
    try:
        payload = get_request_payload()
        record.validate_keys(
            payload,
            schema,
        )
    except (KeyError, TypeError, JSONDecodeError) as err:
        on_error(400, message=err.args[0])
    else:
        return payload

    Journey decorator¶

    campus.flask_campus.journeys ¶

    campus.flask_campus.journeys

    Action-journey decorator for client apps (#828).

    One decorator is the entire API surface app authors touch:

    from campus import flask_campus

@app.post("/assignments/submit")
@flask_campus.journey("campus.submissions.submit")
def submit_assignment(): ...

    @journey names the user action the view serves. Names follow the yapper emission convention (campus.<subject>[.<sub>].<action>) and reuse the same labels (#840): an emission is the event an episode culminates in, so a journey named campus.submissions.submit reads as "the episode that ended in that emission". Mint a new label (same format) only when no emission exists for the action. The journey id itself is adopted from the active request when one exists (the journeys middleware's cookie/header adoption, or a previous hop of the same episode) and minted fresh only when there is none — so decorating a view is always safe, never fragments an ongoing episode, and requires no knowledge of ids, cookies, or headers. fresh=True opts out of adoption for a hard action boundary mid-episode.

    Journeys are observability correlation, not application state: if audit tracing is disabled the decorator still runs but nothing is recorded.

    def journey(
        name: str | None = None,
        *,
        fresh: bool = False,
) -> typing.Callable:
    """Mark a view as part of a (named) user-action journey.

    Args:
        name: Label for the journey, following the yapper emission
            convention (``campus.<subject>[.<sub>].<action>``) and
            reusing the emission label where one exists (#840); lands
            in the span's tags.journey_name for the journey view cards.
        fresh: Start a NEW journey for this request even if one is
            active — a hard action boundary mid-episode. Default False:
            adopt the active journey, minting only when there is none.

    Returns:
        The decorated view.
    """
    def decorator(view):
        @wraps(view)
        def wrapped(*args, **kwargs):
            """Resolve the journey id, stamp g, then run the view."""
            journey_id = getattr(flask.g, "journey_id", None)
            if not journey_id:
                # Standalone use (no journeys middleware): adopt the
                # action cookie, else mint.
                journey_id = (
                    flask.request.cookies.get(
                        ACTION_JOURNEY_COOKIE
                    )
                    or uid.generate_category_uid("journey")
                )
            elif fresh:
                journey_id = uid.generate_category_uid("journey")
            flask.g.journey_id = journey_id
            if name:
                flask.g.journey_name = name
            # Continue the episode after this response (redirects,
            # same-page XHRs) even without the journeys middleware.
            flask.after_this_request(_set_journey_cookie)
            return view(*args, **kwargs)
        return wrapped
    return decorator

    Circle membership helpers¶

    campus.flask_campus.circles ¶

    campus.flask_campus.circles

    Circles-backed membership checks for Flask views.

    This module answers "is the signed-in user in this circle" for view gates (OAuthLoginManager.circle_member_required), resolving membership the same way campus.api's own enforcement does: the caller's user ID (their email, which is also their user circle's name) against the target circle's members. It is view-level gating only — campus.api remains the enforcement point for the resources themselves; this gate exists so apps can render the right UI before the API's 403 wall.

    Checks run as the signed-in user (campus.with_user_session), not with the app's client credentials: client principals hold no domain authority, and circle reads are only open "for now".

    Membership answers are cached process-local per (circle, user) for a short TTL — warm admin views make zero upstream calls. Definitive non-membership (absent member, missing circle) fails closed with False; infrastructure failures raise CirclesUnavailableError so the caller can distinguish a 403 from a degraded backend.

    class MembersCache:
    """Process-local cache of membership answers.

    Keys are (circle_name, user_id) pairs; values are
    (expires_at, is_member). Expired entries are dropped lazily on
    read. Answers only — never the members dict itself: transitive
    resolution (a stated campus.api refinement) changes which users
    count as members, so caching raw membership data would drift.
    """

    def __init__(self, ttl_seconds: float = CIRCLE_MEMBERS_TTL_SECONDS):
        self.ttl_seconds = ttl_seconds
        self._entries: dict[
            tuple[str, str],
            tuple[float, bool],
        ] = {}

    def get(self, circle: str, user_id: str) -> bool | None:
        """The cached answer for (circle, user), or None on miss/expiry."""
        entry = self._entries.get((circle, user_id))
        if entry is None:
            return None
        expires_at, is_member = entry
        if time.monotonic() >= expires_at:
            self._entries.pop((circle, user_id), None)
            return None
        return is_member

    def put(self, circle: str, user_id: str, is_member: bool) -> None:
        """Cache the membership answer for ttl_seconds."""
        self._entries[(circle, user_id)] = (
            time.monotonic() + self.ttl_seconds,
            is_member,
        )

    def is_member(
        campus: _WithUserSession,
        cache: MembersCache,
        circle: str,
        user_id: str,
) -> bool:
    """True if `user_id` is a member of the named circle.

    The circle is resolved by name (names are unique best-effort and
    stable — role circles use dotted names like "timetable.admin";
    user circles are named after the email account, and user IDs are
    emails, so the lookup key is the caller's ID verbatim). A missing
    circle denies fail-closed with a warning — that is a bootstrap
    problem, not a per-user denial.

    Raises:
        CirclesUnavailableError: If the circles backend (or the user
            token flow backing the check) fails to answer.
    """
    cached = cache.get(circle, user_id)
    if cached is not None:
        return cached
    try:
        with campus.with_user_session() as client:
            matches = [
                c for c in client.api.circles.list()
                if c.name == circle
            ]
            if not matches:
                logger.warning(
                    "No circle named '%s' is configured: denying "
                    "membership fail-closed — run the circles "
                    "bootstrap (docs/circles-access-control.md)",
                    circle,
                )
                answer = False
            else:
                # The single-membership probe (campus#1098): the
                # server's answer to "is this principal in this
                # circle" — client checks keep resolving membership in
                # exactly one place when campus.api adds access tiers
                # or transitive resolution. 404 (absent member) is the
                # expected negative answer; other errors surface.
                answer = (
                    client.api.circles[matches[0].id].members.get(user_id)
                    is not None
                )
    except CirclesUnavailableError:
        raise
    except Exception as err:
        logger.error("Circle membership check failed: %s", err)
        raise CirclesUnavailableError(
            f"Could not answer membership in '{circle}': {err}"
        ) from err
    cache.put(circle, user_id, answer)
    return answer

    def reconcile(
        request_args: dict[str, typing.Any],
        func: typing.Callable[..., typing.Any],
        allow_extra: bool = False,
) -> tuple[dict[str, typing.Any], dict[str, typing.Any], list[str]]:
    """Reconcile request arguments with function parameters. Returns a
    tuple of:
    - reconciled arguments (with defaults applied)
    - extra arguments (not in function parameters)
    - missing required parameters

    Args:
        request_args: The arguments from the request (e.g., URL params)
        func: The function whose parameters to reconcile against
        allow_extra: Whether to allow extra arguments not in params
                     if True, they are included in reconciled args
                     if False, they are returned in extra_args
    """
    func_params = dict(inspect.signature(func).parameters)
    MISSING: object = object()
    reconciled: dict[str, typing.Any] = {}
    extra_args: dict[str, typing.Any] = {}
    missing_params: list[str] = []
    for name, param in func_params.items():
        arg = request_args.get(name, MISSING)
        if is_variadic(param):
            # Variadic parameter (*args, **kwargs): only include if
            # present and non-empty
            if arg is not MISSING and arg != {} and arg != ():
                reconciled[name] = arg
        elif is_optional(param):
            # Optional parameter: use default if missing, otherwise use
            # provided value
            reconciled[name] = param.default if arg is MISSING else arg
        else:
            # Required parameter: must be present in request_args
            if arg is MISSING:
                missing_params.append(name)
            else:
                reconciled[name] = arg
    extra_args = {k: v for k, v in request_args.items()
                  if k not in func_params}
    if allow_extra:
        reconciled.update(extra_args)
        extra_args = {}
    return reconciled, extra_args, missing_params

    OpenAPI generator core¶

    campus.flask_campus.openapi ¶

    campus.flask_campus.openapi

    OpenAPI 3 generator for Campus Flask services (campus#1029).

    Each service serves a public, unauthenticated OpenAPI 3 JSON at /{service-prefix}/openapi.json behind campus-gateway. The spec is generated from the Flask app object and committed as a campus/{service}/docs/openapi.json artifact (regenerate with the CLI below; drift and version-pinning unit tests enforce the regenerate-and-bump discipline).

    Two halves, deliberately separable:

    • Schema emitter: functools.singledispatch over campus.common.schema types (Email, DateTime, CampusID, String, Integer, Boolean, Array, Object, ...) and builtin fallbacks, producing plain JSON-Schema dicts. Dispatch lives entirely in the generator — campus.common.schema stays spec-agnostic.
    • App introspector: walks app.url_map (Flask converter → OpenAPI parameter), introspects view signatures (annotations → parameters, defaults → optional), and uses docstrings as the operation documentation: the first line becomes summary and the remainder becomes description. There are no overlay files; the error-envelope components are generated once from the campus.common.errors.api_errors taxonomy (#714).

    Parameter extraction follows the flask_campus.unpack_request signature conventions: GET handlers receive query parameters, all other methods receive a JSON body; path parameters come from the URL rule; a parameter without a default is required. Parameters read from flask.request directly (outside unpack_request) are invisible to the generator.

    This module is the pure generator core: it imports nothing from the services repo, so it moves with the flask-campus carve (#993) untouched. The AppModule-to-app convenience (build_spec_from_module) lives in campus.deploy beside create_app — the legal import direction is service-repo glue importing the consumer layer, never the reverse. The regen entrypoint is scripts/generate_openapi.py in the services repo (generation builds the service app, so it needs that repo's environment).

    Usage::

    from campus.flask_campus import openapi
spec = openapi.generate_spec(
    app, title="Campus Auth API", version="0.2.0",
)
openapi.write_spec(spec, "campus/auth/docs/openapi.json")

    Version pinning: version is explicit and required — the generator never guesses whose release it is pinning. The services repo pins the campus-suite version from its pyproject.toml; a PR that changes the spec must regenerate the artifact and bump that version (minor for spec changes, patch for API-preserving changes).

    Security schemes (bearer/apikey gates) are a phase-2 item and are not emitted yet; response bodies are documented generically (200 + default error envelope).

    def annotation_to_schema(annotation: Any) -> dict[str, Any]:
    """Emit an OpenAPI schema dict for a type annotation.

    Handles the typing constructs that appear in view signatures
    (``X | None`` unions, ``Literal[...]``, ``list[T]``, ``dict[K, V]``)
    and delegates concrete classes to the emitter.
    """
    if annotation is inspect.Parameter.empty or annotation is Any:
        return {}
    if annotation is None or annotation is type(None):
        return {"nullable": True}
    origin = typing.get_origin(annotation)
    if origin is Union or origin is UnionType:
        members = list(typing.get_args(annotation))
        non_none = [m for m in members if m is not type(None)]
        if len(non_none) == 1:
            merged = annotation_to_schema(non_none[0])
            if len(non_none) != len(members):
                merged = dict(merged)
                merged["nullable"] = True
            return merged
        return {"anyOf": [annotation_to_schema(m) for m in members]}
    if origin is Literal:
        literal: dict[str, Any] = {"enum": list(typing.get_args(annotation))}
        value_types = {type(v) for v in typing.get_args(annotation)}
        if value_types and value_types <= {str}:
            literal["type"] = "string"
        elif value_types and all(isinstance(v, bool) for v in typing.get_args(annotation)):
            literal["type"] = "boolean"
        elif value_types and value_types <= {int}:
            literal["type"] = "integer"
        elif value_types and value_types <= {float}:
            literal["type"] = "number"
        return literal
    if origin is list:
        args = typing.get_args(annotation)
        items = annotation_to_schema(args[0]) if args else {}
        return {"type": "array", "items": items}
    if origin is dict:
        args = typing.get_args(annotation)
        obj: dict[str, Any] = {"type": "object"}
        if len(args) == 2 and args[1] is not Any:
            obj["additionalProperties"] = annotation_to_schema(args[1])
        return obj
    if isinstance(annotation, type):
        return _emit_for_class(annotation)
    return {}

    def generate_spec(
        app: flask.Flask,
        *,
        title: str,
        description: str = "",
        version: str,
        exclude_endpoints: Iterable[str] | None = None,
) -> dict[str, Any]:
    """Generate an OpenAPI 3 spec dict from a Flask app object.

    Walks ``app.url_map``; every non-excluded rule becomes a path item,
    every method an operation. ``version`` is required and lands in
    ``info.version`` verbatim — callers pass the release they are
    pinning. The 308 trailing-slash convention is appended to the
    description unless the description already mentions it.
    """
    excluded = (
        DEFAULT_EXCLUDED_ENDPOINTS if exclude_endpoints is None
        else frozenset(exclude_endpoints)
    )
    selected: list[tuple[routing.Rule, str]] = []
    for rule in app.url_map.iter_rules():
        leaf = rule.endpoint.rsplit(".", 1)[-1]
        if leaf in excluded:
            continue
        for method in sorted((rule.methods or set()) - {"HEAD", "OPTIONS"}):
            selected.append((rule, method))

    operation_ids = _operation_ids(selected)

    paths: dict[str, dict[str, Any]] = {}
    tags: set[str] = set()
    for rule, method in selected:
        view = app.view_functions[rule.endpoint]
        operation = _build_operation(rule, method, view)
        operation["operationId"] = operation_ids[(rule.endpoint, method, rule.rule)]
        tag = _endpoint_tag(rule.endpoint)
        if tag:
            operation["tags"] = [tag]
            tags.add(tag)
        path_item = paths.setdefault(_to_openapi_path(rule.rule), {})
        path_item[method.lower()] = operation

    if description and "308" not in description:
        description = f"{description}\n\n{_TRAILING_SLASH_NOTE}"
    elif not description:
        description = _TRAILING_SLASH_NOTE

    error_schemas, error_responses = _error_components()
    return {
        "openapi": "3.0.3",
        "info": {
            "title": title,
            "description": description,
            "version": version,
        },
        "tags": [{"name": tag} for tag in sorted(tags)],
        "paths": paths,
        "components": {
            "schemas": error_schemas,
            "responses": error_responses,
        },
    }
